cmake_minimum_required(VERSION 3.18)
project(zovgame CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)

set(CORE src/core/assets.cpp src/core/world.cpp src/core/game.cpp src/core/settings.cpp src/core/net.cpp src/core/audio.cpp)
set(APP src/app.cpp src/app_render.cpp src/app_ui.cpp src/app_hud.cpp src/gfx/renderer.cpp)

if(ANDROID)
  add_library(zovgame SHARED src/platform/android_main.cpp ${APP} ${CORE})
  target_compile_options(zovgame PRIVATE -O2 -Wall)
  # keep ANativeActivity_onCreate visible (hidden preset above)
  set_target_properties(zovgame PROPERTIES LINK_FLAGS "-Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL")
  target_link_libraries(zovgame android log EGL GLESv3 aaudio)
else()
  # Desktop: logic tests only (no GL / audio device needed)
  add_executable(t_world tests/t_world.cpp ${CORE})
  add_executable(t_game tests/t_game.cpp ${CORE})
  add_executable(t_net tests/t_net.cpp ${CORE})
  add_executable(t_audio tests/t_audio.cpp ${CORE})
  find_package(Threads REQUIRED)
  target_link_libraries(t_net Threads::Threads)
endif()
